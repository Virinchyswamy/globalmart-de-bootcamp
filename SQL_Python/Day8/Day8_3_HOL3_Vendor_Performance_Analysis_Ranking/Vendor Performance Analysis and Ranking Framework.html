SQL Mental Model

Vendor Ranking in SQL vs Python

SQL's window function DENSE_RANK() does exactly what pandas .rank(method='dense') does — rank without gaps.

SQL pipeline

-- Step 1: Zero out negative delays
-- (on-time delivery → delay = 0, not negative)
SELECT *,
  CASE WHEN delay_duration <= 0
       THEN 0
       ELSE delay_duration
  END AS delay_duration
FROM orders;

-- Step 2: Aggregate per vendor + compute rates
SELECT vendor_id,
  AVG(delay_duration) AS avg_delivery_delay,
  100.0 * SUM(orders_returned) / SUM(orders_delivered)
    AS return_pct,
  100.0 * SUM(orders_late)    / SUM(orders_delivered)
    AS delayed_delivery_pct
FROM orders JOIN vendor_performance USING(vendor_id)
GROUP BY vendor_id;

-- Step 3: Rank on each metric (lower = better)
SELECT vendor_id,
  DENSE_RANK() OVER
    (ORDER BY return_pct ASC)  AS rank_return,
  DENSE_RANK() OVER
    (ORDER BY delayed_pct ASC) AS rank_delayed,
  DENSE_RANK() OVER
    (ORDER BY avg_delay ASC)   AS rank_delay
FROM vendor_summary;

Python equivalent

# Step 1: Custom function + .apply()
# CASE WHEN value <= 0 THEN 0 ELSE value END
def calculate_delay(value):
    return 0 if value <= 0 else value

orders['delay_duration'] = (
    orders['delay_duration'].apply(calculate_delay)
)

# Step 2: groupby().agg() + merge + rate calculation
delivery_summary = orders.groupby('vendor_id').agg(
    avg_delivery_delay=('delay_duration', 'mean')
).reset_index()

# Step 3: .rank(method='dense', ascending=True)
# → lowest value gets rank 1 (best performance)
vendor_rank['rank_by_return%'] = (
    vendor_rank['return%']
    .rank(method='dense').astype(int)
)

# Average rank = mean across 3 rank columns
vendor_rank['average_rank'] = vendor_rank[[
    'rank_by_return%',
    'rank_by_delayed_delivery%',
    'rank_by_average_delivery_delay'
]].mean(axis=1)
DENSE_RANK vs RANK: If two vendors tie at rank 2, DENSE_RANK gives both rank 2 and the next gets rank 3. Regular RANK skips — next gets rank 4. pandas method='dense' = SQL DENSE_RANK().
Task 1

Build delivery_performance Table

Apply a custom delay function, aggregate, merge, and compute percentage rates — 5 operations to one clean table.

Step-by-step

# ① Define delay function (CASE WHEN in SQL)
# On-time/early orders have negative delay_duration
# Clamp negatives to 0 — no "negative delay"
def calculate_delay(value):
    if value <= 0:
        return 0
    else:
        return value

# ② Apply to every row in delay_duration column
orders['delay_duration'] = (
    orders['delay_duration'].apply(calculate_delay)
)

# ③ Aggregate per vendor, round to 1 decimal
delivery_summary = (
    orders.groupby('vendor_id')
    .agg(
        avg_delivery_duration=('delivery_duration', 'mean'),
        avg_delivery_delay   =('delay_duration',    'mean')
    ).reset_index()
)
delivery_summary['avg_delivery_duration'] = (
    delivery_summary['avg_delivery_duration'].round(1)
)
delivery_summary['avg_delivery_delay'] = (
    delivery_summary['avg_delivery_delay'].round(1)
)

Merge + rate calculation

# ④ Merge vendor_performance + delivery_summary
# on='vendor_id', how='inner' → only matching vendors
delivery_performance = pd.merge(
    vendor_performance,
    delivery_summary,
    on='vendor_id',
    how='inner'
)

# ⑤ Compute percentage rates
# return% = returned orders / delivered orders × 100
# .round(1) → keep one decimal place
delivery_performance['return%'] = (
    100 * delivery_performance['total_orders_returned']
        / delivery_performance['total_orders_delivered']
).round(1)

delivery_performance['delayed_delivery%'] = (
    100 * delivery_performance['total_orders_delivered_late']
        / delivery_performance['total_orders_delivered']
).round(1)

# ⑥ Keep only the 6 required columns
delivery_performance = delivery_performance[[
    'vendor_id', 'vendor_name', 'return%',
    'delayed_delivery%', 'avg_delivery_duration',
    'avg_delivery_delay'
]]
Result: 4 vendors × 6 columns. VEN04 stands out with avg_delivery_delay = 14.1 vs ~4.8 for the others.
Task 2

Rank Vendors Across 3 Metrics

.rank(method='dense', ascending=True) gives rank 1 to the best (lowest) value. Average all 3 ranks for a composite score.

Ranking logic

# Drop avg_delivery_duration — not needed for ranking
vendor_rank = delivery_performance.drop(
    'avg_delivery_duration', axis=1
)

# .rank(method='dense') — like DENSE_RANK() in SQL
# ascending=True → lowest value = rank 1 (best)
# .astype(int) → convert float ranks to integers
vendor_rank['rank_by_return%'] = (
    vendor_rank['return%']
    .rank(method='dense').astype(int)
)

vendor_rank['rank_by_delayed_delivery%'] = (
    vendor_rank['delayed_delivery%']
    .rank(method='dense').astype(int)
)

vendor_rank['rank_by_average_delivery_delay'] = (
    vendor_rank['avg_delivery_delay']
    .rank(method='dense').astype(int)
)

# Average rank across all 3 dimensions
# axis=1 → compute mean across COLUMNS (per row)
# axis=0 would compute mean across ROWS (per column)
vendor_rank['average_rank'] = vendor_rank[[
    'rank_by_return%',
    'rank_by_delayed_delivery%',
    'rank_by_average_delivery_delay'
]].mean(axis=1)

Full Code

Complete Vendor Ranking Pipeline

import numpy as np
import pandas as pd

# ── Load data ───────────────────────────────────────────
vendor_performance = pd.read_csv("vendor_summary.csv")
orders             = pd.read_csv("order_summary.csv")

# ── Task 1: Build delivery_performance ─────────────────
def calculate_delay(value):
    return 0 if value <= 0 else value  # clamp negatives to 0

orders['delay_duration'] = orders['delay_duration'].apply(calculate_delay)

delivery_summary = (
    orders.groupby('vendor_id').agg(
        avg_delivery_duration=('delivery_duration', 'mean'),
        avg_delivery_delay   =('delay_duration',    'mean')
    ).reset_index()
)
delivery_summary['avg_delivery_duration'] = delivery_summary['avg_delivery_duration'].round(1)
delivery_summary['avg_delivery_delay']    = delivery_summary['avg_delivery_delay'].round(1)

delivery_performance = pd.merge(vendor_performance, delivery_summary,
                                 on='vendor_id', how='inner')
delivery_performance['return%'] = (
    100 * delivery_performance['total_orders_returned']
        / delivery_performance['total_orders_delivered']
).round(1)
delivery_performance['delayed_delivery%'] = (
    100 * delivery_performance['total_orders_delivered_late']
        / delivery_performance['total_orders_delivered']
).round(1)
delivery_performance = delivery_performance[[
    'vendor_id', 'vendor_name', 'return%',
    'delayed_delivery%', 'avg_delivery_duration', 'avg_delivery_delay'
]]

# ── Task 2: Rank vendors ────────────────────────────────
vendor_rank = delivery_performance.drop('avg_delivery_duration', axis=1)

# dense rank ascending: lowest metric = best = rank 1
vendor_rank['rank_by_return%'] = (
    vendor_rank['return%'].rank(method='dense').astype(int))
vendor_rank['rank_by_delayed_delivery%'] = (
    vendor_rank['delayed_delivery%'].rank(method='dense').astype(int))
vendor_rank['rank_by_average_delivery_delay'] = (
    vendor_rank['avg_delivery_delay'].rank(method='dense').astype(int))

# axis=1: average across 3 rank columns per row
vendor_rank['average_rank'] = vendor_rank[[
    'rank_by_return%', 'rank_by_delayed_delivery%',
    'rank_by_average_delivery_delay'
]].mean(axis=1)
